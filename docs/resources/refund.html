Module mercadopago.resources.refund

Module: refund. Payment and refund identifiers are URL-encoded as individual path segments.

from mercadopago.core import MPBase


class Refund(MPBase):
    """Access to payment refunds."""

    def list_all(self, payment_id, request_options=None):
        return self._get(
            uri="/v1/payments/" + self._path_param(payment_id) + "/refunds",
            request_options=request_options,
        )

    def create(self, payment_id, refund_object=None, request_options=None):
        if refund_object is not None and not isinstance(refund_object, dict):
            raise ValueError("Param refund_object must be a Dictionary")
        data = {} if refund_object is None else refund_object
        return self._post(
            uri="/v1/payments/" + self._path_param(payment_id) + "/refunds",
            data=data,
            request_options=request_options,
        )

    def get(self, payment_id, refund_id, request_options=None):
        return self._get(
            uri=f"/v1/payments/{self._path_param(payment_id)}"
                f"/refunds/{self._path_param(refund_id)}",
            request_options=request_options,
        )
